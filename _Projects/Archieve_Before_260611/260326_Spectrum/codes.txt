overlap = 0.5;          % 50%重叠
tapers = 5;             % 5个锥形窗口

fprintf('• 窗口大小: %.1f 秒\n', window_size);

fprintf('• 锥形窗口数: %d\n', tapers);
fprintf('• 频率范围: %.1f-%.1f Hz\n', freq_range);

% 计算时频谱图
[spectrogram_matrix, freq_vector, time_vector] = compute_multitaper_spectrogram_fixed(...
    eeg_signal, Fs, window_size, overlap, tapers, freq_range);

fprintf('✅ 时频谱图计算完成\n');
fprintf('时间点: %d, 频率点: %d\n', length(time_vector), length(freq_vector));

%% ==== 3. 计算整体功率指标 ====
fprintf('\n=== 计算整体功率指标 ===\n');

% 总功率时程 (在指定频率范围内)
freq_mask = freq_vector >= freq_range(1) & freq_vector <= freq_range(2);
total_power = sum(spectrogram_matrix(:, freq_mask), 2);

% 中值频率时程 (反映频谱重心)
median_freq = zeros(length(time_vector), 1);
for i = 1:length(time_vector)
    current_spectrum = spectrogram_matrix(i, freq_mask);
    current_freq = freq_vector(freq_mask);
    cumulative_power = cumsum(current_spectrum);
    total_power_point = cumulative_power(end);
    if total_power_point > 0
        median_idx = find(cumulative_power >= total_power_point/2, 1);
        if ~isempty(median_idx)
            median_freq(i) = current_freq(median_idx);
        end
    end
end

%% ==== 修复的多锥形谱分析函数 ====
function [spectrogram_matrix, freq_vector, time_vector] = compute_multitaper_spectrogram_fixed(signal, Fs, window_size, overlap, tapers, freq_range)
    % 修复版多锥形谱时频谱图计算
    
    window_samples = round(window_size * Fs);
    step_samples = round(window_size * (1-overlap) * Fs);
    
    total_samples = length(signal);
    num_windows = floor((total_samples - window_samples) / step_samples) + 1;
    
    % 设置多锥形谱参数
    params.Fs = Fs;
    params.tapers = [3, 5]; % [时间带宽积, 锥形数] - 更稳定的设置
    params.fpass = freq_range; % 频率范围
    params.pad = 0; % 无填充
    
    fprintf('计算多锥形谱: ');
    
    % 先计算第一个窗口来确定频率向量长度
    test_signal = signal(1:min(window_samples, length(signal)));
    [temp_spectrum, temp_freq] = mtspectrumc(test_signal, params);
    
    freq_vector = temp_freq;
    num_freq_bins = length(freq_vector);
    
    fprintf('频率点数: %d\n', num_freq_bins);
    
    % 初始化输出矩阵 - 现在我们知道确切大小了
    spectrogram_matrix = zeros(num_windows, num_freq_bins);
    time_vector = zeros(1, num_windows);
    
    for i = 1:num_windows
        if mod(i, 20) == 0
            fprintf('%d/%d...', i, num_windows);
        end
        
        start_sample = (i-1) * step_samples + 1;
        end_sample = start_sample + window_samples - 1;
        
        if end_sample > total_samples
            break;
        end
        
        % 提取当前窗口信号
        window_signal = signal(start_sample:end_sample);
        
        % 多锥形谱计算
        try
            [power_spectrum, ~] = mtspectrumc(window_signal, params);
            spectrogram_matrix(i, :) = power_spectrum';
            time_vector(i) = (start_sample + end_sample) / (2 * Fs);
        catch ME
            fprintf('窗口 %d 计算失败: %s\n', i, ME.message);
            % 用前一个窗口的值填充（如果可用）
            if i > 1
                spectrogram_matrix(i, :) = spectrogram_matrix(i-1, :);
            else
                spectrogram_matrix(i, :) = zeros(1, num_freq_bins);
            end
            time_vector(i) = (start_sample + end_sample) / (2 * Fs);
        end
    end
    
    fprintf('完成\n');
    
    % 输出调试信息
    fprintf('频谱矩阵大小: %d x %d\n', size(spectrogram_matrix, 1), size(spectrogram_matrix, 2));
    fprintf('频率向量大小: %d\n', length(freq_vector));
    fprintf('时间向量大小: %d\n', length(time_vector));
end
